<!-- New surface this plan adds -->
```ts
type Domain = 'compass' | 'address' | 'verdicts';

type PromotionState = 'idle' | 'saving' | 'saved' | 'error';

interface UseEvContextPromotionArgs<TPayload, TResult> {
  domain: Domain;
  isLoggedIn: boolean;
  userId: string | null | undefined;
  apiData: unknown; // hook treats falsy / empty array / empty object as "API empty"
  apiWriter: (payload: TPayload) => Promise<TResult>; // app-supplied writer
  enabled?: boolean; // default true; lets caller gate without unmounting
}

interface UseEvContextPromotionReturn<TPayload> {
  shouldPrompt: boolean;
  payload: TPayload | null; // the ev-context guest body the banner will save
  promote: () => Promise<void>; // calls apiWriter, then mirrors via setAuthedSlice + clears the dismissal stamp if any
  dismiss: () => Promise<void>; // stamps promotionDismissed[domain] in authed slice for current userId
  status: PromotionState;
  error: Error | null;
}

function useEvContextPromotion<TPayload, TResult>(
  args: UseEvContextPromotionArgs<TPayload, TResult>
): UseEvContextPromotionReturn<TPayload>;
```

**Step 2 — Build the hook.**

Hook contract (see `<interfaces>` above for full type signature):

```js
import { useEffect, useState, useCallback, useRef } from 'react';
import { evContext } from './evContext.js';

export function useEvContextPromotion({
  domain,             // 'compass' | 'address' | 'verdicts'
  isLoggedIn,
  userId,
  apiData,            // caller passes the API state for the domain; hook treats falsy/empty as "API empty"
  apiWriter,          // async (payload) => result
  enabled = true,
}) {
  // 1. Local state: shouldPrompt, payload, status ('idle'|'saving'|'saved'|'error'), error.
  // 2. On (isLoggedIn, userId, apiData, enabled, domain) change:
  //    - If !enabled || !isLoggedIn || !userId → shouldPrompt=false, payload=null. Return.
  //    - If apiData is non-empty (see isApiEmpty below) → shouldPrompt=false, payload=null. Return.
  //    - Read evContext.getAuthedSlice(userId). If authed.promotionDismissed?.[domain] === true → shouldPrompt=false. Return.
  //    - Read evContext.get(). If guest top-level [domain] is populated (see isGuestPopulated below) → shouldPrompt=true, payload=guest[domain].
  //    - Else shouldPrompt=false, payload=null.
  // 3. Subscribe to evContext.subscribe(fn) so dismiss/promote in another tab updates the local state.
  //    - Re-run the detection logic from step 2 on every broker update.
  //    - Cleanup unsubscribe on unmount.
  // 4. promote(): set status='saving'. Call apiWriter(payload). On success:
  //      a. evContext.setAuthedSlice(userId, { [domain]: payload }) — seeds the mirror cache so the
  //         user gets the same instant SWR hydration on subsequent loads (matches 260426-mc5 semantics).
  //      b. Clear the dismissal flag if set: evContext.setAuthedSlice(userId, { promotionDismissed: { [domain]: false } })
  //         (or call setAuthedFlag if that's the chosen surface).
  //      c. Set status='saved', shouldPrompt=false.
  //    On failure: set status='error', error=err. Leave shouldPrompt as-is so user can retry.
  // 5. dismiss(): write evContext.setAuthedSlice(userId, { promotionDismissed: { [domain]: true } }).
  //    Set shouldPrompt=false locally (don't wait for the subscribe callback).

  // Helper: isApiEmpty(domain, apiData)
  //   - compass: apiData is null/undefined OR (typeof apiData === 'object' && Object.keys(apiData).length === 0) OR (apiData.answers && Object.keys(apiData.answers).length === 0)
  //   - address: apiData is null/undefined/'' OR (object with no formatted/lat/lng)
  //   - verdicts: apiData is null/undefined OR (typeof apiData === 'object' && Object.keys(apiData).length === 0)
  //   Be conservative: when in doubt, treat as NON-empty (don't prompt).

  // Helper: isGuestPopulated(domain, fullEvContext)
  //   - compass: fullEvContext?.compass?.answers && Object.keys(...).length > 0
  //   - address: fullEvContext?.address?.formatted (or whatever shape essentials writes)
  //   - verdicts: fullEvContext?.verdicts && Object.keys(...).length > 0
}
```
