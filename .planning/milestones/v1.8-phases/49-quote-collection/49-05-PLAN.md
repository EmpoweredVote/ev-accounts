import csv
with open('EV-Backend/data/quote_collection.csv') as f:
    reader = csv.reader(f)
    header = next(reader)
    rows = list(reader)

valid_topics = {'healthcare','abortion','tariffs','taxes','same-sex-marriage','religious-freedom','trans-athletes','ukraine-support','medicare','fossil-fuels','voting-rights','deportation','social-security','ai-regulation','climate-change','civil-rights','housing','campaign-finance','immigration','misinformation','redistricting'}

batch_names = ['George Whitesides', 'Laura Friedman', 'Brad Sherman', 'Tony Cardenas', 'Judy Chu', 'Pete Aguilar']
errors = []
for name in batch_names:
    person_rows = [r for r in rows if r[0] == name]
    if len(person_rows) == 0:
        errors.append(f'No rows for {name}')
    for r in person_rows:
        if r[1] not in valid_topics:
            errors.append(f'Invalid topic: {r[1]} for {r[0]}')
        if not r[2].strip():
            errors.append(f'Empty quote for {r[0]} {r[1]}')
        if not r[3].strip().startswith('http'):
            errors.append(f'Bad URL for {r[0]} {r[1]}')
        if not r[4].strip():
            errors.append(f'Empty source_name for {r[0]} {r[1]}')

if errors:
    for e in errors[:10]:
        print(f'FAIL: {e}')
else:
    by_name = {}
    for r in rows:
        by_name.setdefault(r[0], []).append(r)
    for name in sorted(by_name.keys()):
        print(f'{name}: {len(by_name[name])} quotes, {len(set(r[1] for r in by_name[name]))} topics')
    print(f'Total rows: {len(rows)}')
    print(f'Total politicians: {len(by_name)}')
    print('PASS: LA County House batch 1 validation complete')
