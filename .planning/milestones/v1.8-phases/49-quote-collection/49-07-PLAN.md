import csv
from urllib.parse import urlparse

with open('EV-Backend/data/quote_collection.csv') as f:
    reader = csv.reader(f)
    header = next(reader)
    rows = list(reader)

valid_topics = {'healthcare','abortion','tariffs','taxes','same-sex-marriage','religious-freedom','trans-athletes','ukraine-support','medicare','fossil-fuels','voting-rights','deportation','social-security','ai-regulation','climate-change','civil-rights','housing','campaign-finance','immigration','misinformation','redistricting'}

errors = []

# Header
if header != ['full_name','topic_key','quote_text','source_url','source_name']:
    errors.append(f'Bad header: {header}')

# Row-level checks
for i, r in enumerate(rows):
    if len(r) != 5:
        errors.append(f'Row {i+2} has {len(r)} fields')
        continue
    if r[1] not in valid_topics:
        errors.append(f'Invalid topic: {r[1]} for {r[0]}')
    if not r[2].strip():
        errors.append(f'Empty quote_text row {i+2}')
    if not r[3].strip().startswith('http'):
        errors.append(f'Bad URL row {i+2}')
    if not r[4].strip():
        errors.append(f'Empty source_name row {i+2}')

# No general pages
bad_patterns = ['congress.gov/member/', 'congress.gov/bill/', 'congress.gov/vote/', 'roll-call-vote', 'bloomington.in.gov/mayor']
for i, r in enumerate(rows):
    for p in bad_patterns:
        if p in r[3]:
            errors.append(f'General page still present row {i+2}: {p}')

# Duplicates
seen = set()
for r in rows:
    key = (r[0], r[1], r[2][:100])
    if key in seen:
        errors.append(f'Duplicate: {r[0]} {r[1]}')
    seen.add(key)

# Source domains
domains = set()
for r in rows:
    parsed = urlparse(r[3])
    domains.add(parsed.netloc)

if errors:
    for e in errors[:20]:
        print(f'FAIL: {e}')
else:
    print(f'Total rows: {len(rows)}')
    by_name = {}
    for r in rows:
        by_name.setdefault(r[0], []).append(r)
    print(f'Politicians with quotes: {len(by_name)}')
    print(f'Source domains: {sorted(domains)}')
    print()
    print('PASS: Cleaned CSV validation complete — all rows cite specific verifiable sources')
