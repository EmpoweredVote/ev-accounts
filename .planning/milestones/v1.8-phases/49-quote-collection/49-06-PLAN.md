import csv

with open('EV-Backend/data/quote_collection.csv') as f:
    reader = csv.reader(f)
    header = next(reader)
    rows = list(reader)

expected_header = ['full_name','topic_key','quote_text','source_url','source_name']
valid_topics = {'healthcare','abortion','tariffs','taxes','same-sex-marriage','religious-freedom','trans-athletes','ukraine-support','medicare','fossil-fuels','voting-rights','deportation','social-security','ai-regulation','climate-change','civil-rights','housing','campaign-finance','immigration','misinformation','redistricting'}
expected_names = {'Gavin Newsom','Eleni Kounalakis','Mike Braun','Micah Beckwith','Alex Padilla','Adam Schiff','Todd Young','Jim Banks','Erin Houchin','George Whitesides','Laura Friedman','Brad Sherman','Tony Cardenas','Judy Chu','Pete Aguilar','Jimmy Gomez','Ted Lieu','Sydney Kamlager-Dove','Linda Sanchez','Maxine Waters','Nanette Barragan','Karen Bass','Kerry Thomson'}

errors = []

# Header check
if header != expected_header:
    errors.append(f'Bad header: {header}')

# All 23 politicians present
found_names = set(r[0] for r in rows)
missing = expected_names - found_names
if missing:
    errors.append(f'Missing politicians: {missing}')
extra = found_names - expected_names
if extra:
    errors.append(f'Unexpected names: {extra}')

# Row-level checks
for i, r in enumerate(rows):
    if len(r) != 5:
        errors.append(f'Row {i+2} has {len(r)} fields (expected 5)')
        continue
    if r[1] not in valid_topics:
        errors.append(f'Invalid topic: {r[1]} for {r[0]} (row {i+2})')
    if not r[2].strip():
        errors.append(f'Empty quote_text row {i+2}')
    if not r[3].strip().startswith('http'):
        errors.append(f'Bad URL row {i+2}: {r[3][:50]}')
    if not r[4].strip():
        errors.append(f'Empty source_name row {i+2}')

# Duplicate check
seen = set()
for r in rows:
    key = (r[0], r[1], r[2][:100])
    if key in seen:
        errors.append(f'Duplicate: {r[0]} {r[1]}')
    seen.add(key)

if errors:
    for e in errors[:20]:
        print(f'FAIL: {e}')
    print(f'Total errors: {len(errors)}')
else:
    by_name = {}
    for r in rows:
        by_name.setdefault(r[0], []).append(r)
    print('=== PER-POLITICIAN COVERAGE ===')
    for name in sorted(by_name.keys()):
        topics = set(r[1] for r in by_name[name])
        print(f'{name}: {len(by_name[name])} quotes, {len(topics)}/21 topics')
    print(f'\\n=== TOTALS ===')
    print(f'Total rows: {len(rows)}')
    print(f'Total politicians: {len(by_name)}')
    all_topics = {}
    for r in rows:
        all_topics.setdefault(r[1], 0)
        all_topics[r[1]] += 1
    print(f'\\n=== TOPIC COVERAGE ===')
    for topic in sorted(all_topics.keys(), key=lambda t: all_topics[t], reverse=True):
        print(f'{topic}: {all_topics[topic]} quotes')
    print(f'\\nPASS: Final validation complete — {len(rows)} quotes across {len(by_name)} politicians')
