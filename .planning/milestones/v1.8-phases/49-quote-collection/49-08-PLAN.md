import csv, re

with open('EV-Backend/data/quote_collection.csv') as f:
    reader = csv.reader(f)
    header = next(reader)
    rows = list(reader)

# Check for date patterns in URLs
date_pattern = re.compile(r'/20[12]\d[/-]')
dated = []
undated = []
for r in rows:
    if date_pattern.search(r[3]):
        dated.append(r)
    else:
        undated.append(r)

print(f'Total rows: {len(rows)}')
print(f'Rows with date in URL: {len(dated)}')
print(f'Rows without date in URL: {len(undated)}')
if undated:
    print('Undated URLs:')
    for r in undated:
        print(f'  {r[0]} / {r[1]}: {r[3][:80]}')

# Zero general pages
bad = [r for r in rows if any(p in r[3] for p in ['congress.gov/member/','congress.gov/bill/','congress.gov/vote/','bloomington.in.gov/mayor'])]
if bad:
    print(f'FAIL: {len(bad)} general page rows remain')
else:
    pct = len(dated)/len(rows)*100 if rows else 0
    print(f'\\n{pct:.0f}% of rows have extractable dates from source URLs')
    print('PASS: QUOTE-02 date traceability confirmed via source URLs')
