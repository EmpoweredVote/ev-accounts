import csv
with open('EV-Backend/data/quote_collection.csv') as f:
    reader = csv.reader(f)
    header = next(reader)
    rows = list(reader)

expected_header = ['full_name','topic_key','quote_text','source_url','source_name']
valid_topics = {'healthcare','abortion','tariffs','taxes','same-sex-marriage','religious-freedom','trans-athletes','ukraine-support','medicare','fossil-fuels','voting-rights','deportation','social-security','ai-regulation','climate-change','civil-rights','housing','campaign-finance','immigration','misinformation','redistricting'}

errors = []
if header != expected_header:
    errors.append(f'Bad header: {header}')

newsom = [r for r in rows if r[0] == 'Gavin Newsom']
kounalakis = [r for r in rows if r[0] == 'Eleni Kounalakis']

for r in rows:
    if r[1] not in valid_topics:
        errors.append(f'Invalid topic: {r[1]} for {r[0]}')
    if not r[2].strip():
        errors.append(f'Empty quote_text for {r[0]} {r[1]}')
    if not r[3].strip().startswith('http'):
        errors.append(f'Bad URL for {r[0]} {r[1]}: {r[3][:50]}')
    if not r[4].strip():
        errors.append(f'Empty source_name for {r[0]} {r[1]}')

if errors:
    for e in errors[:10]:
        print(f'FAIL: {e}')
else:
    print(f'Newsom quotes: {len(newsom)}')
    print(f'Kounalakis quotes: {len(kounalakis)}')
    print(f'Total rows: {len(rows)}')
    print(f'Topics covered (Newsom): {len(set(r[1] for r in newsom))}')
    print(f'Topics covered (Kounalakis): {len(set(r[1] for r in kounalakis))}')
    print('PASS: CSV validation complete')
