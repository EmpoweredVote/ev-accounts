Quick summary query:
```sql
SELECT
  g.name as city,
  COUNT(DISTINCT p.id) as total_officials,
  COUNT(DISTINCT pa.politician_id) as officials_with_stances,
  COUNT(pa.topic_id) as total_stances
FROM essentials.governments g
JOIN essentials.districts d ON d.geo_id = g.geo_id
JOIN essentials.offices o ON o.district_id = d.id
JOIN essentials.politicians p ON p.id = o.politician_id
LEFT JOIN inform.politician_answers pa ON pa.politician_id = p.id
WHERE g.geo_id IN ('2545560', '2562535', '2539835')
GROUP BY g.name
ORDER BY g.name;
-- Newton: 25 officials, 25 with_stances, >0 total_stances
-- Somerville: 12 officials, 12 with_stances, >0 total_stances
-- Medford: 8 officials, 8 with_stances (or 7 + honest-skip documented), >0 total_stances
```
