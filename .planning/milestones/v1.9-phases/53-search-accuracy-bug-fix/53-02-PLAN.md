**usePoliticianData.js — rewrite to remove warming/retry logic:**

Replace the entire hook implementation. The new version:
1. Removes the ZIP vs address branching — ALL queries go through `searchPoliticians(query)`
2. Removes the warming/retry loop (no more 202 polling)
3. Removes the `maxRetries` option (no longer needed)
4. Keeps the `enabled` and `initialData` options
5. Keeps the AbortController pattern for cancellation
6. Keeps the same return shape: `{ data, phase, error, dataStatus, formattedAddress }`
7. Removes the "warming" phase — only "idle", "loading", "fresh", "error" remain

New implementation:
```javascript
import { useState, useEffect, useRef } from "react";
import { searchPoliticians } from "../lib/api";

const EMPTY_ARRAY = [];
let instanceCounter = 0;

/**
 * Custom hook for fetching politician data.
 *
 * All queries (ZIP codes, cities, addresses) go through the unified
 * POST /essentials/politicians/search endpoint. The backend handles
 * area vs point detection and returns results synchronously.
 *
 * @param {string} query - ZIP code, city name, or address
 * @param {Object} options - Configuration options
 * @param {boolean} options.enabled - Gate to prevent fetching (default: true)
 * @param {Array} options.initialData - Pre-populated data from sessionStorage (default: [])
 *
 * @returns {Object} { data, phase, error, dataStatus, formattedAddress }
 * - data: Array of politicians
 * - phase: "idle" | "loading" | "fresh" | "error"
 * - error: Error message string or null
 * - dataStatus: "fresh" | "no-geofence-data" | null
 * - formattedAddress: Backend-validated formatted address string
 */
export function usePoliticianData(query, options = {}) {
  const {
    enabled = true,
    initialData = EMPTY_ARRAY,
  } = options;

  const [data, setData] = useState(initialData);
  const [phase, setPhase] = useState("idle");
  const [error, setError] = useState(null);
  const [dataStatus, setDataStatus] = useState(null);
  const [formattedAddress, setFormattedAddress] = useState("");

  const controllerRef = useRef(null);

  useEffect(() => {
    if (!enabled || !query) {
      setPhase("idle");
      setFormattedAddress("");
      return;
    }

    // Abort any previous in-flight request
    controllerRef.current?.abort();

    const controller = new AbortController();
    controllerRef.current = controller;

    const id = ++instanceCounter;
    console.log(`[usePoliticianData] #${id} starting for query="${query}"`);

    async function fetchData() {
      try {
        setError(null);
        setFormattedAddress("");
        setData(initialData);
        setPhase("loading");

        const result = await searchPoliticians(query);

        if (controller.signal.aborted) return;

        if (result.error) {
          setError(result.error);
          setPhase("error");
          return;
        }

        setData(Array.isArray(result.data) ? result.data : []);
        setDataStatus(result.status || "fresh");
        setFormattedAddress(result.formattedAddress || "");
        setPhase("fresh");
        console.log(`[usePoliticianData] #${id} complete — ${(result.data || []).length} officials`);
      } catch (err) {
        if (err.name === "AbortError") {
          console.log(`[usePoliticianData] #${id} aborted`);
          return;
        }
        console.log(`[usePoliticianData] #${id} error: ${err.message}`);
        setError(err.message);
        setPhase("error");
      }
    }

    fetchData();

    return () => {
      console.log(`[usePoliticianData] #${id} cleanup — aborting`);
      controller.abort();
    };
  }, [query, enabled]); // eslint-disable-line react-hooks/exhaustive-deps

  return { data, phase, error, dataStatus, formattedAddress };
}
```

**SRCH-02 Root Cause Analysis:**

The re-search bug occurs because:
1. When the user selects a new place from autocomplete on the results page, `onPlaceSelected` fires and sets `hasValidSelection = true`
2. The user clicks "Search", which calls `handleAddressSearch`
3. `handleAddressSearch` clears `cachedResult` and sets `setSearchParams({ q: addressInput })`
4. The `queryFromUrl` derives from `searchParams.get('q')` — this updates reactively
5. BUT the `usePoliticianData` hook has `enabled: !!activeQuery && !cachedResult` — the `cachedResult` was just set to `null` in the same render cycle
6. The real problem: `cachedResult` is cleared via `setCachedResult(null)` and `sessionStorage.removeItem('ev:results')` in `handleAddressSearch`, but if the new `queryFromUrl` equals the old one (e.g., user re-searches same location), the hook doesn't re-trigger because `query` hasn't changed
7. Additionally, Google Places autocomplete may lose its listener attachment after the component re-renders, causing the first attempt to not register the selection
