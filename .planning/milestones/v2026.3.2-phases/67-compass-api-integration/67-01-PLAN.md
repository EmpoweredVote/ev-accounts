```go
func sessionCookie(name, value string, maxAge int) *http.Cookie {
	secure := true
	sameSite := http.SameSiteNoneMode
	domain := ".empowered.vote"

	port := os.Getenv("PORT")
	if port == "" || strings.HasPrefix(port, "5050") {
		secure = false
		sameSite = http.SameSiteLaxMode
		domain = "" // no domain restriction in local dev
	}

	return &http.Cookie{
		Name:     name,
		Value:    value,
		Path:     "/",
		Domain:   domain,
		MaxAge:   maxAge,
		HttpOnly: true,
		Secure:   secure,
		SameSite: sameSite,
	}
}
```

This is the ONLY change to this file. Do NOT modify any handler functions. The Domain field enables the session cookie to be sent from both compass.empowered.vote and essentials.empowered.vote to api.empowered.vote.
