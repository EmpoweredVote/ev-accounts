2. Inside the existing useEffect (after the onAuthStateChange subscription), add an async `initSso()` function that runs ONCE on mount. Call it with `void initSso()` (do NOT await — fire-and-forget inside useEffect).

   `initSso()` logic:
   ```ts
   async function initSso() {
     try {
       const { data: { session: existingSession } } = await supabase.auth.getSession();
       if (existingSession) {
         // Already have a local session — no SSO needed
         return;
       }

       const accountsApiUrl = import.meta.env.VITE_ACCOUNTS_API_URL as string;
       if (!accountsApiUrl) {
         console.error('[SSO] VITE_ACCOUNTS_API_URL not configured');
         return;
       }

       const controller = new AbortController();
       const timeout = setTimeout(() => controller.abort(), 3000);

       const res = await fetch(`${accountsApiUrl}/api/auth/session`, {
         credentials: 'include',
         signal: controller.signal,
       });
       clearTimeout(timeout);

       if (!res.ok) {
         // 401 = no cookie, expected for unauthenticated users — silent fallback
         return;
       }

       const tokens = await res.json();
       if (!tokens.access_token || !tokens.refresh_token) {
         console.error('[SSO] Response missing required tokens');
         return;
       }

       const { error } = await supabase.auth.setSession({
         access_token: tokens.access_token,
         refresh_token: tokens.refresh_token,
       });
       if (error) {
         console.error('[SSO] setSession failed:', error.message);
       }
       // On success, onAuthStateChange fires SIGNED_IN which sets session/user and triggers fetchUserProfile — no manual work needed
     } catch (err) {
       // AbortError (timeout) or network failure — silent fallback to unauthenticated
       if (err instanceof Error && err.name !== 'AbortError') {
         console.error('[SSO] session check failed:', err.message);
       }
     } finally {
       setIsAuthChecking(false);
     }
   }
   ```

3. Add `isAuthChecking` to the Provider value:
   ```tsx
   <AuthContext.Provider value={{ session, user, profile, isLoadingProfile, isAuthChecking, signOut }}>
   ```

CRITICAL anti-patterns to avoid:
- Do NOT set `isAuthChecking = false` inside onAuthStateChange callback — ONLY in initSso()'s finally block
- Do NOT call `fetchUserProfile()` after setSession() — onAuthStateChange SIGNED_IN event handles it
- Do NOT use `apiFetch` from apiClient.ts — it targets wrong URL and doesn't pass credentials: include
- Do NOT use `VITE_API_URL` — that's the VQ backend, not the accounts API

Updated component:
```tsx
export function PrivateRoute() {
  const { session, isAuthChecking } = useAuth();
  const location = useLocation();

  if (isAuthChecking) {
    return null;
  }

  if (!session) {
    return <Navigate to="/login" replace state={{ from: location }} />;
  }

  return <Outlet />;
}
```

This preserves deep links because the URL doesn't change while isAuthChecking is true — once SSO resolves and session is set, the requested route renders directly.
