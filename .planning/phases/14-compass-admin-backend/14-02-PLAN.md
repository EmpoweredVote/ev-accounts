```typescript
// 1. adminCreateTopicWithStances — uses admin_create_topic_with_stances RPC
export async function adminCreateTopicWithStances(data: {
  title: string;
  question_text: string;
  short_title?: string;
  is_live?: boolean;
  stances?: Array<{ value: number; text: string }>;
}): Promise<Record<string, unknown>>
```
Calls adminRpc('admin_create_topic_with_stances', { p_title, p_question_text, p_short_title: data.short_title ?? null, p_is_live: data.is_live ?? false, p_stances: JSON.stringify(data.stances ?? []) }). If error.message starts with 'INVALID_STANCE_VALUE' or 'INVALID_STANCE_TEXT', throw Object.assign(new Error(error.message), { code: 'VALIDATION_ERROR' }). Returns result as Record<string, unknown>.

```typescript
// 2. adminListTopics — direct supabaseAdmin query
export async function adminListTopics(): Promise<Record<string, unknown>[]>
```
supabaseAdmin.schema('inform').from('compass_topics').select('id, title, short_title, is_live, created_at, updated_at').order('created_at', { ascending: false }). Returns rows array. Do NOT select is_active (generated column — avoid selecting it to prevent type confusion). Do NOT join stances — list view returns metadata only per CONTEXT decision.

```typescript
// 3. adminCreatePolitician — direct supabaseAdmin insert
export async function adminCreatePolitician(data: {
  first_name: string;
  last_name: string;
  preferred_name?: string;
  full_name?: string;
  office_title?: string;
  photo_origin_url?: string;
  is_candidate?: boolean;
}): Promise<Record<string, unknown>>
```
supabaseAdmin.schema('inform').from('politicians').insert({ ...data }).select().single(). If error throw new Error(error.message). Returns row.

```typescript
// 4. adminUpdatePolitician — direct supabaseAdmin update
export async function adminUpdatePolitician(
  politicianId: string,
  data: Partial<{
    first_name: string;
    last_name: string;
    preferred_name: string;
    full_name: string;
    office_title: string;
    photo_origin_url: string;
    is_active: boolean;
    is_candidate: boolean;
  }>
): Promise<Record<string, unknown>>
```
supabaseAdmin.schema('inform').from('politicians').update(data).eq('id', politicianId).select().single(). If error.code === 'PGRST116' (no row returned), throw Object.assign(new Error('Politician not found'), { code: 'NOT_FOUND' }). Returns row.

```typescript
// 5. adminListCategories — direct supabaseAdmin query
export async function adminListCategories(): Promise<Record<string, unknown>[]>
```
supabaseAdmin.schema('inform').from('compass_categories').select('id, title, created_at').order('title', { ascending: true }). Returns rows array.

```typescript
// 6. adminCreateCategory — direct supabaseAdmin insert
export async function adminCreateCategory(data: { title: string }): Promise<Record<string, unknown>>
```
supabaseAdmin.schema('inform').from('compass_categories').insert({ title: data.title }).select().single(). If error.code === '23505' (unique constraint), throw Object.assign(new Error('Category title already exists'), { code: 'DUPLICATE_TITLE' }). If other error throw new Error(error.message). Returns row.

```typescript
// 7. adminAssignTopicCategories — uses admin_assign_topic_categories RPC
export async function adminAssignTopicCategories(
  topicId: string,
  categoryIds: string[]
): Promise<void>
```
Calls adminRpc('admin_assign_topic_categories', { p_topic_id: topicId, p_category_ids: JSON.stringify(categoryIds) }). If error.message === 'NOT_FOUND', throw Object.assign(new Error('Topic not found'), { code: 'NOT_FOUND' }). If other error throw new Error(error.message).
