```go
type ImportFederalVotesConfig struct {
    DryRun          bool
    CongressNumbers []int  // default [119, 118]

    LegiScanAPIKey  string // for Senate votes via LegiScan

}

type ImportFederalVotesResult struct {
    HouseVotesUpserted  int
    SenateVotesUpserted int
    LegiScanBridgeRows  int // new id_type="legiscan" bridge rows created
    Skipped             int
    Errors              []string
}
```

**2. ImportFederalVotes main function — flow:**

Step 1: Validate config. Default CongressNumbers to `[]int{119, 118}`. Default MaxErrors to 50.

Step 2: Load bioguide bridge table into `map[string]uuid.UUID` (bioguideID → politicianID).

Step 3: **House Votes Import** (skip if SenateOnly):
  - Create CongressClient: `client := NewCongressClient(cfg.CongressAPIKey)`
  - For each congress in CongressNumbers:
    - Get or create session UUID via `getOrCreateSession(congress)`
    - For each session number (1, 2):
      - Call `client.GetHouseVoteList(ctx, congress, session)`
      - Handle 404/empty gracefully (session 2 may not exist yet for current Congress)
      - For each roll call in the list:
        - Call `client.GetHouseVoteMemberVotes(ctx, congress, session, rollCall.RollCallNumber)`
        - Parse voteDate from rollCall.VoteDate (format "YYYY-MM-DD")
        - Attempt to find matching bill in DB by congress+number if rollCall.BillNumber is non-empty:
          ```go
          var billID *uuid.UUID
          if rollCall.BillNumber != "" {
              // Try to find the bill — normalize number for lookup
              var bill LegislativeBill
              externalID := fmt.Sprintf("%d-%s", congress, strings.ReplaceAll(rollCall.BillNumber, " ", "-"))
              if err := db.DB.Where("external_id = ? AND jurisdiction = ?", externalID, "federal").First(&bill).Error; err == nil {
                  billID = &bill.ID
              }
          }
          ```
        - For each member vote, look up politician via bioguide bridge:
          ```go
          polID, ok := bioguideMap[memberVote.BioguideID]
          if !ok { continue } // skip non-cached politicians
          ```
        - Upsert LegislativeVote:
          - ExternalVoteID: `fmt.Sprintf("house-%d-%d-%d", congress, session, rollCall.RollCallNumber)`
          - Position: `normalizeVoteCast(memberVote.VoteCast)`
          - Result: `normalizeVoteResult(rollCall.Result)`
          - VoteQuestion: rollCall.VoteQuestion
          - VoteDate: parsed date
          - BillID: billID (nullable)
          - YeaCount: rollCall.YeaTotal
          - NayCount: rollCall.NayTotal
          - Source: "congress"
          - SessionID: sessionID
          - `clause.OnConflict{Columns: [{politician_id}, {bill_id}, {session_id}, {external_vote_id}], DoUpdates: AssignmentColumns(["position", "vote_question", "result"])}`
      - Log: `"House votes for %dth Congress session %d: %d roll calls processed"`

Step 4: **Senate Votes Import** (skip if HouseOnly):
  - Create LegiScanClient: `lsClient, err := NewLegiScanClient(cfg.LegiScanAPIKey, "")`
  - **Budget check:** Call `lsClient.RemainingBudget()`. Log: `"LegiScan budget remaining: %d/%d queries"`. If remaining < 1000, log warning: `"WARNING: LegiScan budget low — Senate import may exhaust monthly budget"`.

  a. **Build LegiScan people_id bridge** (one-time per session):
    - For each congress, call `lsClient.Query(ctx, "getSessionList", {"state": "US"})` to get session list
    - Parse response to find session_id matching the congress number (match by year_start: 119th Congress = 2025, 118th = 2023)
    - Call `lsClient.Query(ctx, "getSessionPeople", {"id": strconv.Itoa(legiScanSessionID)})` to get all legislators
    - Parse response to get `[]LegiScanPerson`
    - For each person where Role = "Sen" (senators only — House is handled by Congress.gov):
      - Try matching to existing politician by name: query `essentials.politicians WHERE LOWER(last_name) = LOWER(?) AND LOWER(first_name) = LOWER(?)` — only accept SINGLE match (skip ambiguous)
      - If matched, check if `legislative_politician_id_map` already has `id_type="legiscan"` for this politician. If not, INSERT new bridge row:
        ```go
        bridge := LegislativePoliticianIDMap{
            PoliticianID: matchedPolID,
            IDType:       "legiscan",
            IDValue:      strconv.Itoa(person.PeopleID),
            VerifiedAt:   time.Now(),
            Source:       "legiscan-session-people",
        }
        db.DB.Clauses(clause.OnConflict{DoNothing: true}).Create(&bridge)
        ```
    - Build `legiscanMap`: `map[int]uuid.UUID` mapping people_id → politician UUID
    - Log: `"Built LegiScan bridge: %d senators matched"`

  b. **Import Senate votes via LegiScan:**
    - For each legiScanSessionID (current and previous):
      - Call `lsClient.Query(ctx, "getMasterList", {"id": strconv.Itoa(legiScanSessionID)})` to get bill list
      - **CRITICAL: getMasterList returns a MAP, not an array:**
        ```go
        var wrapper struct {
            Status     string                     `json:"status"`
            MasterList map[string]json.RawMessage `json:"masterlist"`
        }
        json.Unmarshal(body, &wrapper)
        ```
      - Skip key "0" (session metadata)
      - Parse each value as:
        ```go
        type masterListBill struct {
            BillID int    `json:"bill_id"`
            Number string `json:"number"`
            Title  string `json:"title"`
        }
        ```
      - Sort bill_ids for deterministic processing order (enables resume from DB state)
      - For each bill_id:
        - Check if votes already exist for this bill in the DB (for resumability):
          ```go
          var existingCount int64
          db.DB.Model(&LegislativeVote{}).Where("source = ? AND external_vote_id LIKE ?", "legiscan", fmt.Sprintf("legiscan-%%")).
              Where("bill_id IN (SELECT id FROM essentials.legislative_bills WHERE external_id LIKE ? AND jurisdiction = 'federal')", fmt.Sprintf("%%-%s", number)).
              Count(&existingCount)
          ```
          (This is an optimization — skip bills that already have votes imported. Can be simplified to just upsert everything if too complex.)
        - Call `lsClient.Query(ctx, "getBill", {"id": strconv.Itoa(billID)})` to get bill with votes array
        - Parse response:
          ```go
          var billResp struct {
              Bill struct {
                  BillID int `json:"bill_id"`
                  Votes  []struct {
                      RollCallID int    `json:"roll_call_id"`
                      Date       string `json:"date"`
                      Desc       string `json:"desc"`
                      Yea        int    `json:"yea"`
                      Nay        int    `json:"nay"`
                      Passed     int    `json:"passed"`
                      Chamber    string `json:"chamber"` // "S" or "H"
                  } `json:"votes"`
              } `json:"bill"`
          }
          ```
        - For each vote where Chamber == "S" (Senate only — skip House):
          - Call `lsClient.Query(ctx, "getRollCall", {"id": strconv.Itoa(rollCallID)})`
          - Parse response into LegiScanRollCall struct
          - For each individual vote in rollCall.Votes:
            - Look up politician UUID from legiscanMap[vote.PeopleID]
            - If not found, skip (not a senator we have in our DB)
            - Attempt to find matching bill in DB:
              ```go
              var dbBill LegislativeBill
              // Try to find by LegiScan bill number mapped to our external_id format
              // LegiScan number format varies (e.g., "S 123", "SB 456")
              // Best effort: search by number LIKE pattern
              ```
            - Upsert LegislativeVote:
              - ExternalVoteID: `fmt.Sprintf("legiscan-%d", rollCallID)`
              - Position: `normalizeVoteCast(vote.VoteText)`
              - Result: if rollCall.Passed == 1 then "passed" else "failed"
              - VoteQuestion: rollCall.Desc
              - VoteDate: parse rollCall.Date
              - BillID: nullable (linked if found)
              - YeaCount: rollCall.Yea
              - NayCount: rollCall.Nay
              - Source: "legiscan"
              - SessionID: sessionID (from getOrCreateSession for the congress number)
        - Check LegiScan budget periodically (every 100 bills): if remaining < 100, log warning and stop Senate import early
      - Log: `"Senate votes for session %d: %d bills processed, %d roll calls"`

**CRITICAL anti-patterns to avoid:**
- Do NOT use LegiScan for House votes — House = Congress.gov ONLY, Senate = LegiScan ONLY (Pitfall 4 from research)

- Senate external_vote_id format: `"legiscan-{roll_call_id}"`
- normalizeVoteCast handles BOTH Congress.gov values ("Aye"/"Not Voting") AND LegiScan values ("Yea"/"NV")

```go
case "import-federal-votes":
    dryRun := false
    houseOnly := false
    senateOnly := false
    congresses := []int{119, 118}
    maxErrors := 50
    for _, arg := range os.Args[2:] {
        switch {
        case arg == "--dry-run":
            dryRun = true
        case arg == "--house-only":
            houseOnly = true
        case arg == "--senate-only":
            senateOnly = true
        case strings.HasPrefix(arg, "--congress="):
            n, err := strconv.Atoi(strings.TrimPrefix(arg, "--congress="))
            if err != nil {
                log.Fatal("invalid --congress value: ", arg)
            }
            congresses = []int{n}
        case strings.HasPrefix(arg, "--max-errors="):
            n, err := strconv.Atoi(strings.TrimPrefix(arg, "--max-errors="))
            if err != nil {
                log.Fatal("invalid --max-errors value: ", arg)
            }
            maxErrors = n
        }
    }
    congressAPIKey := os.Getenv("CONGRESS_API_KEY")
    legiScanAPIKey := os.Getenv("LEGISCAN_API_KEY")
    if !senateOnly && congressAPIKey == "" {
        log.Fatal("CONGRESS_API_KEY environment variable is required for House votes")
    }
    if !houseOnly && legiScanAPIKey == "" {
        log.Fatal("LEGISCAN_API_KEY environment variable is required for Senate votes")
    }
    result, err := essentials.ImportFederalVotes(essentials.ImportFederalVotesConfig{
        DryRun:          dryRun,
        CongressNumbers: congresses,
        CongressAPIKey:  congressAPIKey,
        LegiScanAPIKey:  legiScanAPIKey,
        HouseOnly:       houseOnly,
        SenateOnly:      senateOnly,
        MaxErrors:       maxErrors,
    })
    if err != nil {
        log.Fatal("import-federal-votes failed: ", err)
    }
    fmt.Printf("Import complete: %d House votes, %d Senate votes, %d LegiScan bridge rows, %d skipped\n",
        result.HouseVotesUpserted, result.SenateVotesUpserted, result.LegiScanBridgeRows, result.Skipped)
    if len(result.Errors) > 0 {
        fmt.Printf("Errors (%d):\n", len(result.Errors))
        for _, e := range result.Errors {
            fmt.Printf("  - %s\n", e)
        }
    }
    os.Exit(0)
```

Env vars:
