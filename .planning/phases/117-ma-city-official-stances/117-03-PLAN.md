-- Final gate: all 7 cities have stances
SELECT g.city,
       COUNT(DISTINCT pa.politician_id) as officials_with_stances,
       COUNT(pa.topic_id) as total_stances,
       (SELECT COUNT(*) FROM essentials.politicians p2
        JOIN essentials.offices o2 ON o2.politician_id=p2.id
        JOIN essentials.chambers ch2 ON ch2.id=o2.chamber_id
        JOIN essentials.governments g2 ON g2.id=ch2.government_id
        WHERE g2.id=g.id) as total_officials
FROM inform.politician_answers pa
JOIN essentials.politicians p ON p.id = pa.politician_id
JOIN essentials.offices o ON o.politician_id = p.id
JOIN essentials.chambers ch ON ch.id = o.chamber_id
JOIN essentials.governments g ON g.id = ch.government_id
WHERE g.city IN ('Boston','Cambridge','Worcester','Springfield','Lowell','Brockton','Quincy')
  AND g.state = 'MA'
GROUP BY g.city, g.id
ORDER BY g.city;
-- Expected: 7 rows, officials_with_stances > 0 for every city

-- Global sourcing gate (no unsourced context anywhere in phase)
SELECT COUNT(*) AS total_uncited
FROM inform.politician_context pc
JOIN essentials.politicians p ON p.id = pc.politician_id
JOIN essentials.offices o ON o.politician_id = p.id
JOIN essentials.chambers ch ON ch.id = o.chamber_id
JOIN essentials.governments g ON g.id = ch.government_id
WHERE g.city IN ('Boston','Cambridge','Worcester','Springfield','Lowell','Brockton','Quincy')
  AND g.state = 'MA'
  AND (pc.sources IS NULL OR array_length(pc.sources, 1) IS NULL OR array_length(pc.sources, 1) = 0);

JOIN essentials.chambers ch ON ch.id = o.chamber_id
JOIN essentials.governments g ON g.id = ch.government_id
WHERE g.city IN ('Boston','Cambridge','Worcester','Springfield','Lowell','Brockton','Quincy')
  AND g.state = 'MA'
  AND pc.politician_id IS NULL;
