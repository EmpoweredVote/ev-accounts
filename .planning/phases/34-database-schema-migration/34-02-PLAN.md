Write three migration files following the exact pattern from `20260226000016_inform_rls_grants.sql`:

**For each migration file, use this structure:**
```sql
BEGIN;

-- =============================================================================
-- Migration NNN: RLS policies and grants for the {schema} schema
-- =============================================================================
-- {table count} tables. {Policy rationale}.
-- NO INSERT/UPDATE/DELETE policies — all writes via service role (pool.query()).
-- =============================================================================

-- Section 1: Enable RLS
ALTER TABLE {schema}.{table1} ENABLE ROW LEVEL SECURITY;
ALTER TABLE {schema}.{table2} ENABLE ROW LEVEL SECURITY;
-- ... all tables

-- Section 2: Public-read policies (reference tables)
CREATE POLICY "{table1}: public read"
  ON {schema}.{table1}
  FOR SELECT
  TO anon, authenticated
  USING (true);
-- ... repeat for each reference table

-- Section 3: Owner-read policies (user-linked tables, if any per 34-01 findings)
-- Only if 34-01-SUMMARY identified user_id columns in this schema
CREATE POLICY "{table}: owner select"
  ON {schema}.{table}
  FOR SELECT
  TO authenticated
  USING ((select auth.uid()) = user_id);

-- Section 4: Grants
GRANT USAGE ON SCHEMA {schema} TO anon, authenticated;
GRANT SELECT ON ALL TABLES IN SCHEMA {schema} TO anon, authenticated;
ALTER DEFAULT PRIVILEGES IN SCHEMA {schema} GRANT SELECT ON TABLES TO anon, authenticated;

COMMIT;
```
