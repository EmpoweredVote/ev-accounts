```typescript
export async function runCalibrationLapseJob(): Promise<void> {
  const today = new Date().toISOString().slice(0, 10); // 'YYYY-MM-DD'
  const jobStart = Date.now();

  // CRON-04: Idempotency — abort if already ran today
  const { rowCount } = await pool.query(
    'INSERT INTO public.calibration_lapse_runs (run_date) VALUES ($1) ON CONFLICT (run_date) DO NOTHING',
    [today]
  );
  if (rowCount === 0) {
    console.log(`[cron] Calibration lapse already ran for ${today} — skipping`);
    return;
  }

  let warned25 = 0, warned30 = 0, demoted = 0;

  try {
    // Fetch all three thresholds
    // get_calibration_lapsed_users returns users who have NOT calibrated topics
    // that went live >= p_days_threshold days ago
    const { data: day25Users } = await supabaseAdmin.rpc('get_calibration_lapsed_users', { p_days_threshold: 25 });
    const { data: day30Users } = await supabaseAdmin.rpc('get_calibration_lapsed_users', { p_days_threshold: 30 });
    const { data: day31Users } = await supabaseAdmin.rpc('get_calibration_lapsed_users', { p_days_threshold: 31 });

    // Build sets for deduplication: day31 users should NOT also get day25/day30 notifications
    // day30 users should NOT also get day25 notifications
    const day31Set = new Set((day31Users ?? []).map((u: { user_id: string }) => u.user_id));
    const day30Set = new Set((day30Users ?? []).map((u: { user_id: string }) => u.user_id));

    // Day 31: Demotion (highest priority — process first)
    for (const user of (day31Users ?? [])) {
      try {
        await executeDemotion(user.user_id, {
          reason: 'calibration_lapse',
          overdue_topics: user.overdue_topic_ids,
          days_overdue: user.days_overdue,
          triggered_by: 'cron'
        });
        // Write demotion confirmation notification
        await pool.query(
          `INSERT INTO public.notifications (user_id, type, payload) VALUES ($1, $2, $3)`,
          [user.user_id, 'demotion_confirmed', JSON.stringify({
            reason: 'calibration_lapse',
            overdue_topic_ids: user.overdue_topic_ids,
            days_overdue: user.days_overdue,
            message: 'Your Empowered status has been revoked due to uncalibrated topics. Complete calibration and re-empower to restore your status.'
          })]
        );
        demoted++;
      } catch (err) {
        console.error(`[cron] Failed to demote user ${user.user_id}:`, err);
        // Continue processing other users — don't let one failure stop the batch
      }
    }

    // Day 30: Final warning (exclude users already demoted at day 31)
    for (const user of (day30Users ?? [])) {
      if (day31Set.has(user.user_id)) continue; // already demoted
      await pool.query(
        `INSERT INTO public.notifications (user_id, type, payload) VALUES ($1, $2, $3)`,
        [user.user_id, 'calibration_warning_30', JSON.stringify({
          overdue_topic_ids: user.overdue_topic_ids,
          days_overdue: user.days_overdue,
          message: 'Final warning: calibrate your outstanding topics before the next daily check or your Empowered status will be revoked.'
        })]
      );
      warned30++;
    }

    // Day 25: Warning (exclude users at day 30+ — they get a more urgent message)
    for (const user of (day25Users ?? [])) {
      if (day30Set.has(user.user_id)) continue; // already warned at 30 level
      if (day31Set.has(user.user_id)) continue; // already demoted
      await pool.query(
        `INSERT INTO public.notifications (user_id, type, payload) VALUES ($1, $2, $3)`,
        [user.user_id, 'calibration_warning_25', JSON.stringify({
          overdue_topic_ids: user.overdue_topic_ids,
          days_overdue: user.days_overdue,
          message: 'You have uncalibrated topics that require attention. Calibrate within the next 6 days to maintain your Empowered status.'
        })]
      );
      warned25++;
    }

    // Update run record with results
    await pool.query(
      `UPDATE public.calibration_lapse_runs
       SET finished_at = now(), users_warned_25 = $2, users_warned_30 = $3, users_demoted = $4
       WHERE run_date = $1`,
      [today, warned25, warned30, demoted]
    );
  } catch (err) {
    // Record error but leave finished_at NULL (indicates incomplete run)
    await pool.query(
      'UPDATE public.calibration_lapse_runs SET error_message = $2 WHERE run_date = $1',
      [today, String(err)]
    );
    console.error('[cron] Calibration lapse job failed:', err);
    throw err; // Re-throw so node-cron logs the failure
  }

  console.log(JSON.stringify({
    level: 'info',
    job: 'calibration-lapse',
    run_date: today,
    warned25,
    warned30,
    demoted,
    duration_ms: Date.now() - jobStart
  }));
}
```

Key design decisions:
- Process day 31 FIRST (demotion), then day 30, then day 25 — ensures most severe action happens even if job fails partway
- Deduplication via Sets: a user at day 31 does NOT also get a day-25 or day-30 notification; a user at day 30 does NOT also get a day-25 notification
- Individual try/catch around each demotion — one user failure doesn't block others
- Idempotency: INSERT ON CONFLICT DO NOTHING on run_date (CRON-04)
- finished_at stays NULL if job errors (distinguishes incomplete from complete runs)

```typescript
import cron from 'node-cron';
import { runCalibrationLapseJob } from '../lib/cronService.js';

export function startCalibrationLapseCron(): void {
  cron.schedule('0 2 * * *', async () => {
    try {
      await runCalibrationLapseJob();
    } catch (err) {
      // Error already logged in cronService — prevent unhandled rejection
      console.error('[cron] Unhandled error in calibration lapse job:', err);
    }
  }, {
    timezone: 'UTC',
    name: 'calibration-lapse'
  });
  console.log('[cron] Calibration lapse job registered (daily 02:00 UTC)');
}
```

Note on noOverlap: node-cron v4 uses `noOverlap` to prevent concurrent runs. HOWEVER, the primary idempotency mechanism is the calibration_lapse_runs table INSERT. Use noOverlap if the option is available in the installed version (check node-cron types). If the types don't include noOverlap, omit it — the table-based idempotency is sufficient.
