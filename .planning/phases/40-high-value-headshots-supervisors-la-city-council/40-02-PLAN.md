```python
python3 -c "
import psycopg2, psycopg2.extras, os, sys
from pathlib import Path
sys.path.insert(0, str(Path('EV-Backend/scripts')))
from utils import load_env
load_env()
conn = psycopg2.connect(os.environ['DATABASE_URL'])
cur = conn.cursor(cursor_factory=psycopg2.extras.RealDictCursor)

# Check 1: All photos point to Supabase CDN
cur.execute(\"\"\"
    SELECT p.full_name, pi.url, pi.photo_license, pi.type
    FROM essentials.politician_images pi
    JOIN essentials.politicians p ON p.id = pi.politician_id
    WHERE pi.url LIKE '%supabase.co%'
      AND pi.photo_license IS NOT NULL
    ORDER BY p.full_name
\"\"\")
rows = cur.fetchall()
print(f'Headshots with CDN URL + license: {len(rows)}')
for r in rows:
    print(f'  {r[\"full_name\"]}: {r[\"url\"][:60]}... (license={r[\"photo_license\"]})')

# Check 2: No government domain hotlinks for these 20 officials
cur.execute(\"\"\"
    SELECT p.full_name, pi.url
    FROM essentials.politician_images pi
    JOIN essentials.politicians p ON p.id = pi.politician_id
    JOIN essentials.offices o ON o.politician_id = p.id
    WHERE (LOWER(o.title) LIKE '%supervisor%' OR LOWER(o.title) LIKE '%council%')
      AND pi.url NOT LIKE '%supabase.co%'
      AND pi.type = 'default'
\"\"\")
hotlinks = cur.fetchall()
if hotlinks:
    print(f'WARNING: {len(hotlinks)} government hotlinks remain:')
    for r in hotlinks:
        print(f'  {r[\"full_name\"]}: {r[\"url\"]}')
else:
    print('PASS: No government domain hotlinks for supervisors/council')

# Check 3: No duplicate rows
cur.execute(\"\"\"
    SELECT p.full_name, COUNT(*) as cnt
    FROM essentials.politician_images pi
    JOIN essentials.politicians p ON p.id = pi.politician_id
    WHERE pi.type = 'default'
    GROUP BY p.full_name
    HAVING COUNT(*) > 1
\"\"\")
dupes = cur.fetchall()
if dupes:
    print(f'WARNING: {len(dupes)} politicians with duplicate default images')
    for r in dupes:
        print(f'  {r[\"full_name\"]}: {r[\"cnt\"]} rows')
else:
    print('PASS: No duplicate default image rows')

conn.close()
"
```
