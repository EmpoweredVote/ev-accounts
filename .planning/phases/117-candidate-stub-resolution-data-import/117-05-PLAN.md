    ```typescript
    // Phase 117: Supabase Storage photo upload helper.
    // Ported from EV-Backend/scripts/upload_monroe_council_photos.py.
    // Used by scripts/import-monroe-stub-candidates.ts (Plan 06).
    //
    // Bucket: politician_photos
    // Path:   monroe_2026/{slug}.{ext}
    // Upsert: true (re-run safe — stable public URL across overwrites)

    import { createClient } from '@supabase/supabase-js';

    const BUCKET = 'politician_photos';
    const STORAGE_PREFIX = 'monroe_2026';

    const EXT_BY_TYPE: Record<string, string> = {
      'image/jpeg': 'jpg',
      'image/png': 'png',
      'image/webp': 'webp',
    };

    let _client: ReturnType<typeof createClient> | null = null;

    function getSupabase() {
      if (_client) return _client;
      const url = process.env.SUPABASE_URL;
      const key = process.env.SUPABASE_SERVICE_ROLE_KEY;
      if (!url) throw new Error('uploadCandidatePhoto: SUPABASE_URL not set');
      if (!key) throw new Error('uploadCandidatePhoto: SUPABASE_SERVICE_ROLE_KEY not set');
      // NOTE: do NOT log url/key values here — only their absence
      _client = createClient(url, key);
      return _client;
    }

    export async function uploadCandidatePhoto(
      sourceUrl: string,
      slug: string,
      contentType: 'image/jpeg' | 'image/png' | 'image/webp' = 'image/jpeg'
    ): Promise<string> {
      if (!sourceUrl) throw new Error(`uploadCandidatePhoto(${slug}): sourceUrl empty`);
      if (!slug) throw new Error('uploadCandidatePhoto: slug empty');

      const ext = EXT_BY_TYPE[contentType] ?? 'jpg';
      const storagePath = `${STORAGE_PREFIX}/${slug}.${ext}`;

      // 1. Download source
      const resp = await fetch(sourceUrl);
      if (!resp.ok) {
        throw new Error(`uploadCandidatePhoto(${slug}): download failed ${resp.status} for ${sourceUrl}`);
      }
      const bytes = new Uint8Array(await resp.arrayBuffer());

      // 2. Upload to Supabase Storage
      const supabase = getSupabase();
      const { error: upErr } = await supabase.storage
        .from(BUCKET)
        .upload(storagePath, bytes, { contentType, upsert: true });
      if (upErr) {
        // upErr.message is safe to surface; do NOT dump the full error object
        throw new Error(`uploadCandidatePhoto(${slug}): storage upload failed — ${upErr.message}`);
      }

      // 3. Get public URL
      const { data } = supabase.storage.from(BUCKET).getPublicUrl(storagePath);
      if (!data?.publicUrl) {
        throw new Error(`uploadCandidatePhoto(${slug}): getPublicUrl returned no URL`);
      }
      return data.publicUrl;
    }
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|---|---|
| external source URL → fetch() | Downloading a third-party URL; response bytes land in memory and then Storage |
| env var → Supabase client | SUPABASE_SERVICE_ROLE_KEY read from .env, never logged |
| helper → Supabase Storage bucket | Authenticated write to politician_photos bucket |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|---|---|---|---|---|
| T-117-15 | Information disclosure | SUPABASE_SERVICE_ROLE_KEY leaked via log | mitigate | Grep assertion in Task 1 acceptance rejects any `console.*` call referencing the env var. Error messages surface key ABSENCE only, never the value. |
| T-117-16 | Tampering | attacker-controlled sourceUrl uploads malicious bytes | mitigate | contentType is caller-supplied (from feasibility CSV) — Plan 06 will only pass image/* types. Supabase Storage enforces content-type on the bucket. Downstream render path treats photos as `<img src>` (no script execution surface). |
| T-117-17 | DoS | fetch() on a massive file exhausts memory | accept | Candidate photos from Monroe sources are typically <500KB; full set is ~30 files. Deferred size cap to Plan 06 (the consumer) if it becomes a problem. |
| T-117-18 | Spoofing | wrong politician's photo uploaded due to slug collision | mitigate | Slug is derived from canonical full_name in feasibility CSV. Plan 06's transaction-per-candidate will fail loudly if slug collides with an existing upload from a different candidate. |
</threat_model>
