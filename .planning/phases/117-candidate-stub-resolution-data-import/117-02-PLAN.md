    ```sql
    -- Migration 068: add is_candidate + bio_source_url columns, backfill is_candidate from race_candidates
    --
    -- Phase 117 schema change. Closes the address-leakage bug (memory note
    -- project_candidates_vs_politicians.md) by introducing an explicit is_candidate
    -- flag that the address resolver filters out. Before this migration, the only
    -- candidate/incumbent signal was the overloaded is_incumbent column on
    -- essentials.politicians, which led to non-incumbent candidate rows leaking
    -- into tier-grouped "these are your reps" address lookups.
    --
    -- D-08: is_candidate BOOLEAN NOT NULL DEFAULT false
    -- D-10: backfill is_candidate=true for rows linked via race_candidates.is_incumbent=false
    -- D-14 + Claude's Discretion: bio_source_url TEXT (single-source bio citation)
    --
    -- Both columns and the backfill run in one file, matching the 064/067 precedent.
    BEGIN;

    ALTER TABLE essentials.politicians
      ADD COLUMN IF NOT EXISTS is_candidate BOOLEAN NOT NULL DEFAULT false;

    ALTER TABLE essentials.politicians
      ADD COLUMN IF NOT EXISTS bio_source_url TEXT;

    COMMENT ON COLUMN essentials.politicians.is_candidate IS
      'Phase 117: true = active candidate; false = sitting official. Address resolver (essentialsService.ts) filters is_candidate=true OUT of tier-grouped results. Compass picker and race listings are NOT filtered. An incumbent running for re-election keeps is_candidate=false because they currently hold the seat.';

    COMMENT ON COLUMN essentials.politicians.bio_source_url IS
      'Phase 117: canonical source URL for bio_text. Single-source-only per D-07; never a synthesis of multiple pages. Used by the Phase 117 Monroe County stub import.';

    -- D-10 backfill: any politician linked to a race as non-incumbent is, by definition,
    -- an active candidate rather than a sitting official. This closes the pre-existing
    -- leak in the same pass that introduces the new column.
    UPDATE essentials.politicians p
    SET is_candidate = true
    WHERE EXISTS (
      SELECT 1 FROM essentials.race_candidates rc
      WHERE rc.politician_id = p.id
        AND rc.is_incumbent = false
    );

    -- Logged verification: how many rows are is_candidate=true after backfill
    DO $$
    DECLARE
      v_count int;
    BEGIN
      SELECT COUNT(*) INTO v_count
      FROM essentials.politicians
      WHERE is_candidate = true;
      RAISE NOTICE 'migration 068: is_candidate=true count after backfill: %', v_count;
    END $$;

    -- Pitfall #1 sanity check: warn if any row ended up with BOTH flags true.
    -- This is a warning, not an error — sitting officials running for re-election
    -- should remain is_candidate=false, but we want loud notice if the backfill
    -- caught something unexpected.
    DO $$
    DECLARE
      v_conflict int;
    BEGIN
      SELECT COUNT(*) INTO v_conflict
      FROM essentials.politicians
      WHERE is_candidate = true AND is_incumbent = true;
      IF v_conflict > 0 THEN
        RAISE WARNING 'migration 068: % rows have both is_candidate=true AND is_incumbent=true — review manually', v_conflict;
      END IF;
    END $$;

    COMMIT;
    ```

    Do NOT modify any other file in this task. Do NOT apply the migration yet — Task 2 does that.
