    Create from the PATTERNS.md body, with one critical modification: because `is_candidate` is NOT exposed in the API response (confirmed in PATTERNS.md note), the script must cross-check returned politician IDs against a direct DB query instead of asserting on a `.is_candidate` field.

    Body:
    ```bash
    #!/usr/bin/env bash
    # Phase 117: verify address-lookup leak is closed.
    # Calls the address resolver for a known Monroe County address and asserts
    # NO returned politician has is_candidate=true in the DB (cross-check, since
    # the API does not currently expose is_candidate in its JSON shape).
    set -euo pipefail

    API="${API_URL:-https://api.empowered.vote}"
    ADDRESS="${1:-401 N Morton St, Bloomington, IN 47404}"
    DATABASE_URL="${DATABASE_URL:?DATABASE_URL required}"

    echo "Checking leak closure at $API for: $ADDRESS"

    response=$(curl -fsS --get "$API/api/essentials/representatives" --data-urlencode "address=$ADDRESS")

    # Extract politician IDs from the response (shape: .politicians[].id)
    ids=$(echo "$response" | jq -r '.politicians[]?.id // empty' | sort -u)

    if [[ -z "$ids" ]]; then
      echo "WARN: no politicians returned for $ADDRESS — cannot validate leak"
      exit 0
    fi

    # Build a SQL IN-list and query the DB for any is_candidate=true rows among them
    id_list=$(echo "$ids" | awk '{printf "%s'\''%s'\''", (NR>1?",":""), $0}')
    leaked=$(psql "$DATABASE_URL" -tAc "SELECT full_name FROM essentials.politicians WHERE id IN ($id_list) AND is_candidate = true")

    if [[ -n "$leaked" ]]; then
      echo "FAIL: leaked candidates in address lookup:"
      echo "$leaked"
      exit 1
    fi

    echo "PASS: no is_candidate=true rows in address lookup for $ADDRESS"
    ```

    Make executable:

    ```bash
    #!/usr/bin/env bash
    # Phase 117: verify every imported candidate's profile page loads (200 OK).
    set -euo pipefail

    BASE="${FRONTEND_URL:-https://essentials.empowered.vote}"
    DATABASE_URL="${DATABASE_URL:?DATABASE_URL required}"

    slugs=$(psql "$DATABASE_URL" -Atc "
      SELECT slug FROM essentials.politicians
      WHERE source = 'phase_117_monroe_stub_import' AND slug IS NOT NULL
    ")

    fail=0
    while IFS= read -r slug; do
      [[ -z "$slug" ]] && continue
      url="$BASE/politician/$slug"
      code=$(curl -fsS -o /dev/null -w "%{http_code}" "$url" || echo "000")
      if [[ "$code" == "200" ]]; then
        echo "PASS: $url"
      else
        echo "FAIL: $url ($code)"
        fail=1
      fi
    done <<<"$slugs"

    exit $fail
    ```
