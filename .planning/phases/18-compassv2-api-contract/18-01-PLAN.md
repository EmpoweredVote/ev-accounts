  DROP CONSTRAINT IF EXISTS compass_responses_value_check;

  ADD CONSTRAINT compass_responses_value_check CHECK (value >= 0.5 AND value <= 5.5);

**Section 4: New migrate_guest_compass_state RPC**

```sql
CREATE OR REPLACE FUNCTION public.migrate_guest_compass_state(
  p_user_id   UUID,
  p_answers   JSONB,   -- array of {topic_id: uuid, value: numeric, write_in_text?: text}
  p_selected_topics UUID[] DEFAULT NULL
)
RETURNS void
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $$
DECLARE
  v_answer     JSONB;
  v_topic_id   UUID;
  v_value      NUMERIC;
  v_write_in   TEXT;
BEGIN
  -- Migrate answers: one row per element in p_answers array
  -- ON CONFLICT DO NOTHING: if user already has an answer for this topic, skip (don't overwrite post-signup activity)
  -- Per-row EXCEPTION for FK violations: skip rows referencing non-existent topics (don't abort whole migration)
  FOR v_answer IN SELECT * FROM jsonb_array_elements(p_answers)
  LOOP
    BEGIN
      v_topic_id := (v_answer->>'topic_id')::UUID;
      v_value    := (v_answer->>'value')::NUMERIC;
      v_write_in := v_answer->>'write_in_text';  -- NULL if absent

      INSERT INTO inform.compass_responses (user_id, topic_id, value, write_in_text, visibility, inverted, updated_at)
      VALUES (p_user_id, v_topic_id, v_value, v_write_in, 'private', false, NOW())
      ON CONFLICT (user_id, topic_id) DO NOTHING;

    EXCEPTION WHEN foreign_key_violation THEN
      -- Topic does not exist — skip this answer, continue migration
      NULL;
    END;
  END LOOP;

  -- Migrate selected topics: only if p_selected_topics is non-null and non-empty
  -- ON CONFLICT DO NOTHING: if user already has a connected_profiles row, skip (don't wipe existing selection)
  IF p_selected_topics IS NOT NULL AND array_length(p_selected_topics, 1) > 0 THEN
    UPDATE connect.connected_profiles
    SET selected_topic_ids = p_selected_topics,
        updated_at = NOW()
    WHERE user_id = p_user_id
      AND (selected_topic_ids IS NULL OR selected_topic_ids = '{}');
    -- Only update if no topics already selected — don't overwrite post-signup activity
  END IF;
END;
$$;
```

**Idempotency note:** The ALTER COLUMN TYPE and DROP/ADD CONSTRAINT are safe to re-run on a schema that already has the correct type (Postgres will cast INT→NUMERIC losslessly). The CREATE OR REPLACE functions are inherently idempotent.
