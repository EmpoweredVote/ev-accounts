**Important ordering note:** Express matches routes in registration order. The `/me/jurisdiction` route MUST be registered BEFORE the `/me` route or Express will try to match `/me` with parameter `:id` = `jurisdiction`. Add the jurisdiction route before the existing `router.get('/me', ...)` definition.

**New route to append (after the PATCH /me handler, before any further routes):**

```typescript
// ---------------------------------------------------------------------------
// GET /api/account/me/jurisdiction
// Middleware: requireAuth + requireConnected
//   requireAuth:      Validates JWT, sets req.userId and req.accessToken
//   requireConnected: Blocks Inform-tier users — jurisdiction requires Connected+
//
// Returns 403 if location_consent is false or null — no distinction between
// "never set" and "revoked" (consent revocation is deferred to post-Alpha).
// Returns jurisdiction JSON from resolve_user_jurisdiction RPC when consent is true.
// ---------------------------------------------------------------------------

router.get('/me/jurisdiction', requireAuth, requireConnected, async (req, res: Response) => {
  const authReq = req as AuthenticatedRequest;

  try {
    // Check location_consent — use adminRpc's underlying supabaseAdmin for a fast lookup.
    // This is a trusted server-side check (not a user-facing data read), consistent with
    // how tierGuards.ts uses supabaseAdmin for standing checks.
    const { data: profile, error: profileError } = await supabaseAdmin
      .schema('connect')
      .from('connected_profiles')
      .select('location_consent')
      .eq('user_id', authReq.userId)
      .maybeSingle();

    if (profileError || !profile) {
      res.status(403).json({ code: 'NOT_CONNECTED', error: 'Connected account required' });
      return;
    }

    if (!profile.location_consent) {
      res.status(403).json({
        code: 'LOCATION_CONSENT_REQUIRED',
        error: 'Location must be set before jurisdiction can be retrieved',
      });
      return;
    }

    // Resolve jurisdiction from encrypted coordinates via SECURITY DEFINER RPC
    const { data: jurisdictionData, error: jurisdictionError } = await adminRpc('resolve_user_jurisdiction', {
      p_user_id: authReq.userId,
    }, 'connect');

    if (jurisdictionError) {
      console.error('[GET /api/account/me/jurisdiction] resolve_user_jurisdiction error:', jurisdictionError.message);
      res.status(500).json({ code: 'INTERNAL_ERROR', message: 'An unexpected error occurred' });
      return;
    }

    // Transform internal GEOID keys to user-facing district label keys
    // resolve_user_jurisdiction returns: { congressional, state_senate, state_house, county, school_district }
    const j = (jurisdictionData ?? {}) as Record<string, string | null>;

    res.status(200).json({
      jurisdiction: {
        congressional_district: j.congressional ?? null,
        state_senate_district: j.state_senate ?? null,
        state_house_district: j.state_house ?? null,
        county: j.county ?? null,
        school_district: j.school_district ?? null,
      },
    });
  } catch (err) {
    console.error('[GET /api/account/me/jurisdiction] Unexpected error:', err);
    res.status(500).json({ code: 'INTERNAL_ERROR', message: 'An unexpected error occurred' });
  }
});
```
