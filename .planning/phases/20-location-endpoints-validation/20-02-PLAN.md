**Coverage check helper (add as a module-level function in connect.ts):**

The coverage check runs AFTER geocoding succeeds. It determines if the returned coordinates fall within Indiana or LA County bounding boxes. This is a fast client-side pre-filter before the PostGIS query — if coordinates are obviously outside coverage, reject immediately without calling the RPC.

Indiana bounding box (approximate): lat 37.77–41.76, lng -88.10 to -84.78
LA County bounding box (approximate): lat 33.70–34.82, lng -118.95 to -117.65

```typescript
function isInCoverage(lat: number, lng: number): boolean {
  // Indiana
  if (lat >= 37.77 && lat <= 41.76 && lng >= -88.10 && lng <= -84.78) return true;
  // LA County, California
  if (lat >= 33.70 && lat <= 34.82 && lng >= -118.95 && lng <= -117.65) return true;
  return false;
}
```

**Route handler — append before `export default router;`:**

```typescript
// ---------------------------------------------------------------------------
// POST /api/connect/set-location
// Middleware: requireAuth + requireConnected
//   requireAuth:      Validates JWT, sets req.userId and req.accessToken
//   requireConnected: Blocks Inform-tier users — location requires Connected+
//
// Flow:
//   1. Zod validates request body (address string, min 1 char)
//   2. geocodeAddress() rejects PO Box addresses before any API call
//   3. Google Maps Geocoding API returns { lat, lng } (or throws GeocodingError)
//   4. isInCoverage() bounding-box check — reject out-of-coverage coordinates
//   5. adminRpc upsert_user_location — encrypts coordinates, sets location_consent = true
//   6. adminRpc resolve_user_jurisdiction — returns jurisdiction jsonb
//   7. Transform jurisdiction keys to user-facing names and return response
//
// Coordinates are NEVER logged, returned, or stored outside the RPC call.
// ---------------------------------------------------------------------------

router.post('/set-location', requireAuth, requireConnected, async (req: Request, res: Response): Promise<void> => {
  const { userId } = req as AuthenticatedRequest;

  const parsed = setLocationBodySchema.safeParse(req.body);
  if (!parsed.success) {
    const firstIssue = parsed.error.issues[0];
    res.status(422).json({
      code: 'VALIDATION_ERROR',
      message: firstIssue?.message ?? 'Invalid request body',
    });
    return;
  }

  const { address } = parsed.data;
  // Address is used only for geocoding — discarded immediately after

  let lat: number;
  let lng: number;

  try {
    const coords = await geocodeAddress(address);
    lat = coords.lat;
    lng = coords.lng;
    // address string is no longer referenced after this point
  } catch (err) {
    if (err instanceof GeocodingError) {
      if (err.code === 'PO_BOX_REJECTED') {
        res.status(422).json({ code: 'PO_BOX_REJECTED', message: err.message });
        return;
      }
      if (err.code === 'ADDRESS_NOT_FOUND' || err.code === 'LOW_CONFIDENCE') {
        res.status(422).json({ code: 'ADDRESS_NOT_FOUND', message: err.message });
        return;
      }
      // GEOCODING_API_ERROR — treat as internal error
      console.error('[connect/set-location] Geocoding API error:', err.message);
      res.status(500).json({ code: 'INTERNAL_ERROR', message: 'An unexpected error occurred' });
      return;
    }
    throw err;
  }

  // Coverage check — after geocoding, before any DB write
  if (!isInCoverage(lat, lng)) {
    res.status(422).json({
      code: 'OUT_OF_COVERAGE',
      message: "Your address is outside our current coverage area. We're expanding soon.",
    });
    return;
  }

  try {
    // Encrypt and store coordinates, set location_consent = true
    const { error: upsertError } = await adminRpc('upsert_user_location', {
      p_user_id: userId,
      p_lat: lat,
      p_lng: lng,
    }, 'connect');

    if (upsertError) {
      console.error('[connect/set-location] upsert_user_location error:', upsertError.message);
      res.status(500).json({ code: 'INTERNAL_ERROR', message: 'An unexpected error occurred' });
      return;
    }

    // Resolve jurisdiction — coordinates are now encrypted in DB; RPC decrypts internally
    const { data: jurisdictionData, error: jurisdictionError } = await adminRpc('resolve_user_jurisdiction', {
      p_user_id: userId,
    }, 'connect');

    if (jurisdictionError) {
      console.error('[connect/set-location] resolve_user_jurisdiction error:', jurisdictionError.message);
      res.status(500).json({ code: 'INTERNAL_ERROR', message: 'An unexpected error occurred' });
      return;
    }

    // Transform internal GEOID keys to user-facing district label keys
    // resolve_user_jurisdiction returns: { congressional, state_senate, state_house, county, school_district }
    // Response format: { congressional_district, state_senate_district, state_house_district, county, school_district }
    const j = (jurisdictionData ?? {}) as Record<string, string | null>;

    res.status(200).json({
      location_consent: true,
      jurisdiction: {
        congressional_district: j.congressional ?? null,
        state_senate_district: j.state_senate ?? null,
        state_house_district: j.state_house ?? null,
        county: j.county ?? null,
        school_district: j.school_district ?? null,
      },
    });
  } catch (err) {
    console.error('[connect/set-location] Unexpected error:', err);
    res.status(500).json({ code: 'INTERNAL_ERROR', message: 'An unexpected error occurred' });
  }
});
```
