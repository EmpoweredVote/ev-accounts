4. **After session creation**, process guest_state if present:
```go
if req.GuestState != nil {
    // Bulk insert answers in a transaction
    if len(req.GuestState.Answers) > 0 {
        tx := db.DB.Begin()
        for _, ga := range req.GuestState.Answers {
            tid, err := uuid.Parse(ga.TopicID)
            if err != nil {
                continue // skip invalid UUIDs silently
            }
            a := compass.Answer{
                ID:          utils.GenerateUUID(),
                UserID:      user.UserID,
                TopicID:     tid,
                Value:       ga.Value,
                WriteInText: ga.WriteInText,
            }
            if err := tx.Create(&a).Error; err != nil {
                tx.Rollback()
                // Log but don't fail registration
                log.Printf("Failed to insert guest answer for topic %s: %v", ga.TopicID, err)
                break
            }
        }
        tx.Commit()
    }

    // Upsert selected topics if provided
    if len(req.GuestState.SelectedTopics) > 0 {
        uc := compass.UserCompass{
            UserID:   user.UserID,
            TopicIDs: pq.StringArray(req.GuestState.SelectedTopics),
        }
        db.DB.Create(&uc)
    }
}
```

```go
func DeleteMyAnswersHandler(w http.ResponseWriter, r *http.Request) {
    userID, ok := utils.GetUserIDFromContext(r.Context())
    if !ok {
        http.Error(w, "Unauthorized", http.StatusUnauthorized)
        return
    }

    tx := db.DB.Begin()
    if err := tx.Where("user_id = ?", userID).Delete(&Answer{}).Error; err != nil {
        tx.Rollback()
        http.Error(w, "Failed to clear answers", http.StatusInternalServerError)
        return
    }
    if err := tx.Where("user_id = ?", userID).Delete(&UserCompass{}).Error; err != nil {
        tx.Rollback()
        http.Error(w, "Failed to clear selected topics", http.StatusInternalServerError)
        return
    }
    tx.Commit()
    w.WriteHeader(http.StatusOK)
    fmt.Fprintln(w, "Compass data cleared")
}
```
